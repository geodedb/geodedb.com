{{- $site := .Site -}} {{- $page := . -}} {{/* ═══════════════════════════════════════════════════════════════ GIT-BASED DATE HANDLING Uses GitInfo when available for accurate modification dates. ═══════════════════════════════════════════════════════════════ */}} {{/* Published date - use frontmatter date, fallback to git author date */}} {{- $publishedDate := "" -}} {{- if not .Date.IsZero -}} {{- $publishedDate = .Date.Format "2006-01-02T15:04:05Z07:00" -}} {{- else if .GitInfo -}} {{- $publishedDate = .GitInfo.AuthorDate.Format "2006-01-02T15:04:05Z07:00" -}} {{- end -}} {{/* Modified date - prefer GitInfo (most accurate), then Lastmod */}} {{- $modifiedDate := "" -}} {{- if .GitInfo -}} {{- $modifiedDate = .GitInfo.AuthorDate.Format "2006-01-02T15:04:05Z07:00" -}} {{- else if not .Lastmod.IsZero -}} {{- $modifiedDate = .Lastmod.Format "2006-01-02T15:04:05Z07:00" -}} {{- else if not .Date.IsZero -}} {{- $modifiedDate = .Date.Format "2006-01-02T15:04:05Z07:00" -}} {{- end -}} {{- $schemaData := dict -}} {{/* Base Organization Schema */}} {{- $organization := dict "@type" "Organization" "name" $site.Title "url" $site.BaseURL "logo" (dict "@type" "ImageObject" "url" ($site.Params.contact.email | default "") ) -}} {{- if $site.Params.social_links -}} {{- $socialUrls := slice -}} {{- range $site.Params.social_links -}} {{- if and .username .url -}} {{- $socialUrls = $socialUrls | append (printf .url .username) -}} {{- else if .url -}} {{- $socialUrls = $socialUrls | append .url -}} {{- end -}} {{- end -}} {{- $organization = merge $organization (dict "sameAs" $socialUrls) -}} {{- end -}} {{/* Website Schema */}} {{- $website := dict "@type" "WebSite" "name" $site.Title "url" $site.BaseURL "description" $site.Params.defaultDescription "inLanguage" else if .IsPage -}} {{/* Article/Page Schema */}} {{- $author := dict "@type" "Person" "name" (default $site.Params.author .Params.author) -}} {{/* Build article with git-based dates */}} {{- $article := dict "@type" (cond (eq .Type "blog") "BlogPosting" "Article") "headline" .Title "description" (.Description | default .Summary | plainify | truncate 155) "url" .Permalink "author" $author $publishedDate -}} {{- $article = merge $article (dict "datePublished" $publishedDate) -}} {{- end -}} {{- if $modifiedDate -}} {{- $article = merge $article (dict "dateModified" $modifiedDate) -}} {{- end -}} {{- with .Params.image -}} {{- $article = merge $article (dict "image" (dict "@type" "ImageObject" "url" (. | absURL))) -}} {{- end -}} {{- if .GetTerms "tags" -}} {{- $keywords := slice -}} {{- range .GetTerms "tags" -}} {{- $keywords = $keywords | append .Title -}} {{- end -}} {{- $article = merge $article (dict "keywords" $keywords) -}} {{- end -}} {{- $breadcrumbList := dict "@type" "BreadcrumbList" "itemListElement" (slice) -}} {{- $breadcrumbs := slice -}} {{- $breadcrumbs = $breadcrumbs | append (dict "@type" "ListItem" "position" 1 "name" "Home" "item" $site.BaseURL ) -}} {{- if not .IsHome -}} {{- $breadcrumbs = $breadcrumbs | append (dict "@type" "ListItem" "position" 2 "name" .Title "item" .Permalink ) -}} {{- end -}} {{- $breadcrumbList = merge $breadcrumbList (dict "itemListElement" $breadcrumbs) -}} {{- $schemaData = dict "@context" "https://schema.org" "@graph" (slice $organization $website $article $breadcrumbList) -}} {{- else if .Data.Singular -}} {{/* Taxonomy Schema */}} {{- $collectionPage := dict "@type" "CollectionPage" "name" .Title "description" (.Description | default .Summary | plainify | truncate 155) "url" .Permalink "inLanguage" $webPage := dict "@type" "WebPage" "name" .Title "description" (.Description | default .Summary | plainify | truncate 155) "url" .Permalink "inLanguage" .Language.Lang "isPartOf" $website -}} {{- $schemaData = dict "@context" "https://schema.org" "@graph" (slice $organization $website $webPage) -}} {{- end -}} {{/* Final emit with double-encoding guard: - If $schemaData is already a JSON
